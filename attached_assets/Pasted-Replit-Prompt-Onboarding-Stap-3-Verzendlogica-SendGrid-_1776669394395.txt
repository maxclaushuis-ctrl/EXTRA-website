Replit Prompt — Onboarding Stap 3: Verzendlogica & SendGrid Integratie
Wat je gaat doen
Bouw de volledige verzendinfrastructuur voor onboarding e-mails. Dit omvat de HTML-generatie vanuit templates, het meesturen van PDF-bijlagen via SendGrid, personalisatie per medewerker en de logging van alle verzendingen. Na deze stap is de volledige onboarding module operationeel en vervangt het Mailblue.

javascriptasync function haalBijlagenOp(templateId) {
  const koppelingen = await getBijlagenVoorTemplate(templateId);

  return Promise.all(koppelingen.map(async (koppeling) => {
    const bijlage = koppeling.bijlage;
    const bestandsdata = await fs.readFile(bijlage.bestandspad);

    return {
      naam: bijlage.naam,
      bestandsnaam: bijlage.bestandsnaam,
      content: bestandsdata.toString('base64'),
      type: 'application/pdf',
      disposition: 'attachment',
      bestandsgrootte: bijlage.bestandsgrootte,
    };
  }));
}
controleerBijlagenGrootte(bijlagen)
javascriptfunction controleerBijlagenGrootte(bijlagen) {
  const totaalBytes = bijlagen.reduce((sum, b) => sum + b.bestandsgrootte, 0);
  const maxBytes = 25 * 1024 * 1024; // 25MB

  if (totaalBytes > maxBytes) {
    return {
      geldig: false,
      melding: `Bijlagen zijn te groot (${(totaalBytes / 1024 / 1024).toFixed(1)} MB).
                Maximum is 25 MB. Verwijder bijlagen of verklein de bestanden.`
    };
  }
  return { geldig: true };
}
verstuurOnboardingMail(medewerkerId, templateId)
javascriptasync function verstuurOnboardingMail(medewerkerId, templateId) {
  // 1. Haal medewerker op
  const medewerker = await getMedewerker(medewerkerId);
  if (!medewerker?.email) {
    throw new Error('Medewerker heeft geen e-mailadres');
  }

  // 2. Haal template op
  const template = await getOnboardingTemplate(templateId);
  if (!template) {
    throw new Error('Template niet gevonden');
  }

  // 3. Controleer of template inhoud heeft
  if (!template.content) {
    throw new Error(`Template '${template.naam}' heeft nog geen e-mailinhoud.
                    Maak de template eerst op via de e-mail builder.`);
  }

  // 4. Genereer HTML en plain text
  const html = await genereerOnboardingHTML(template, medewerker);
  const plainText = await genereerOnboardingPlainText(template, medewerker);

  // 5. Haal bijlagen op en controleer grootte
  const bijlagen = await haalBijlagenOp(templateId);
  const grootteCheck = controleerBijlagenGrootte(bijlagen);
  if (!grootteCheck.geldig) {
    throw new Error(grootteCheck.melding);
  }

  // 6. Vervang tags in onderwerpregel
  const onderwerp = vervangTags(template.onderwerp, medewerker);

  // 7. Bouw SendGrid bericht
  const bericht = {
    to: medewerker.email,
    from: {
      email: process.env.SENDGRID_FROM_EMAIL,  // max@doehetextra.nl
      name: process.env.SENDGRID_FROM_NAME,    // Max van EXTRA
    },
    subject: onderwerp,
    html: html,
    text: plainText,
    attachments: bijlagen,  // PDF bijlagen als echte e-mailbijlagen
  };

  // 8. Verstuur via SendGrid
  await sgMail.send(bericht);

  // 9. Log de verzending
  await logOnboardingVerzending({
    medewerker_id: medewerkerId,
    template_id: templateId,
    status: 'verzonden',
    verstuurd_door: 'handmatig',
  });

  // 10. Update medewerker record
  await updateMedewerker(medewerkerId, {
    onboarding_verstuurd: 1,
    onboarding_verstuurd_op: new Date(),
    onboarding_template_id: templateId,
  });

  return { success: true };
}

2. API endpoints
POST /api/medewerkers/:id/onboarding-versturen
javascriptrouter.post('/:id/onboarding-versturen', async (req, res) => {
  const { id } = req.params;
  const { template_id } = req.body;

  try {
    if (!template_id) {
      return res.status(400).json({ error: 'Geen template opgegeven' });
    }

    await verstuurOnboardingMail(id, template_id);

    res.json({
      success: true,
      verstuurd_op: new Date(),
      bericht: 'Onboarding mail succesvol verstuurd'
    });

  } catch (err) {
    await logOnboardingVerzending({
      medewerker_id: id,
      template_id,
      status: 'mislukt',
      fout_melding: err.message,
    });

    res.status(500).json({
      error: 'Versturen mislukt',
      details: err.message
    });
  }
});
POST /api/medewerkers/onboarding-bulk
javascript// Body: { medewerker_ids: [1, 2, 3] }
// Verwerkt sequentieel met 300ms pauze tussen verzendingen

router.post('/onboarding-bulk', async (req, res) => {
  const { medewerker_ids } = req.body;
  const resultaten = { verzonden: 0, mislukt: 0, fouten: [] };

  for (const id of medewerker_ids) {
    try {
      const medewerker = await getMedewerker(id);
      const template = await selecteerOnboardingTemplate(medewerker);

      if (!template) {
        resultaten.mislukt++;
        resultaten.fouten.push({ id, fout: 'Geen passende template gevonden' });
        continue;
      }

      await verstuurOnboardingMail(id, template.id);
      resultaten.verzonden++;

      await new Promise(r => setTimeout(r, 300));

    } catch (err) {
      resultaten.mislukt++;
      resultaten.fouten.push({ id, fout: err.message });
    }
  }

  res.json(resultaten);
});
