## 2. Werkdagen & tijdvenster berekening

Schrijf een utility module `schedulerUtils.js` met de volgende functies:

### `isWerkdag(datum)`
```javascript
// Geeft true als de datum een werkdag is (ma-vr)
// datum = JavaScript Date object
// Gebruik getDay(): 0 = zondag, 6 = zaterdag
function isWerkdag(datum) {
  const dag = datum.getDay();
  return dag !== 0 && dag !== 6;
}
```

### `isBinnenTijdvenster(datum, start, eind)`
```javascript
// Geeft true als het tijdstip binnen het ingestelde venster valt
// start en eind zijn strings in 'HH:MM' formaat
// Vergelijk uren en minuten van de datum
function isBinnenTijdvenster(datum, start, eind) {
  const uren = datum.getHours();
  const minuten = datum.getMinutes();
  const [startUur, startMin] = start.split(':').map(Number);
  const [eindUur, eindMin] = eind.split(':').map(Number);
  
  const huidigMinuten = uren * 60 + minuten;
  const startMinuten = startUur * 60 + startMin;
  const eindMinuten = eindUur * 60 + eindMin;
  
  return huidigMinuten >= startMinuten && huidigMinuten <= eindMinuten;
}
```

### `berekenWerkelijkVerzendMoment(gewenstMoment, alleenWerkdagen, tijdvensterStart, tijdvensterEind)`
```javascript
// Berekent het daadwerkelijke verzendmoment op basis van:
// - Het gewenste moment
// - Of alleen werkdagen toegestaan zijn
// - Het tijdvenster

// Logica:
// 1. Start met het gewenste moment
// 2. Als alleen_werkdagen = true EN het is geen werkdag:
//    → Verschuif naar de eerstvolgende maandag (behoud tijd)
// 3. Als het tijdstip vóór tijdvenster_start valt:
//    → Verschuif tijd naar tijdvenster_start van dezelfde dag
//    → Als dat een niet-werkdag is, ga naar volgende werkdag
// 4. Als het tijdstip ná tijdvenster_eind valt:
//    → Verschuif naar tijdvenster_start van de VOLGENDE dag
//    → Als die dag een niet-werkdag is, ga naar volgende werkdag
// 5. Geef het gecorrigeerde moment terug

// Voorbeeld:
// Gewenst: vrijdag 20:00, tijdvenster 08:00-18:00, alleen werkdagen
// Resultaat: maandag 08:00
```

## 3. Hoofdscheduler — uitbreiding en consolidatie

Consolideer alle bestaande schedulers (flow scheduler uit stap 5, A/B checker uit stap 6) en voeg de campagne-verzendscheduler toe in één centrale `scheduler.js`:

```javascript
// scheduler.js — Centrale scheduler voor het EXTRA e-mailplatform

import { runFlowScheduler } from './flowEngine.js';
import { checkABWinners } from './abService.js';
import { checkGeplandeCampagnes } from './campaignScheduler.js';
import { logScheduler } from './schedulerUtils.js';

export async function runAllSchedulers() {
  const startTijd = Date.now();
  
  try {
    // 1. Check geplande campagnes (bulk)
    await checkGeplandeCampagnes();
    
    // 2. Verwerk flow stappen
    await runFlowScheduler();
    
    // 3. Check A/B winnaars
    await checkABWinners();
    
    const duur = Date.now() - startTijd;
    await logScheduler('scheduler_run', null, `Voltooid in ${duur}ms`);
    
  } catch (err) {
    console.error('Scheduler fout:', err);
    await logScheduler('scheduler_error', null, err.message);
  }
}

// Start de scheduler elke 5 minuten
setInterval(runAllSchedulers, 5 * 60 * 1000);

// Direct uitvoeren bij server start
runAllSchedulers();
```

## 4. Campagne verzendscheduler

Schrijf `campaignScheduler.js`:

### `checkGeplandeCampagnes()`
```javascript
// Wordt aangeroepen vanuit de hoofdscheduler

async function checkGeplandeCampagnes() {
  // Haal alle campagnes op waarbij:
  // - status = 'gepland'
  // - verzend_direct = 0
  // - werkelijk_verzend_op <= NOW()
  
  for (const campagne of campagnes) {
    try {
      await logScheduler('verzending_gestart', campagne.id, 
        `Starten verzending: ${campagne.naam}`);
      
      // Update status naar 'actief'
      await updateCampagneStatus(campagne.id, 'actief');
      
      // Start verzending
      const resultaat = await sendCampaignBatch(campagne.id);
      
      // Update status naar 'voltooid'
      await updateCampagneStatus(campagne.id, 'voltooid');
      
      // Maak notificatie aan (stap 6 notificatiesysteem)
      await maakNotificatie({
        type: 'campagne_voltooid',
        titel: `Campagne verzonden: ${campagne.naam}`,
        bericht: `Succesvol verzonden naar ${resultaat.verzonden} contacten. 
                  ${resultaat.mislukt > 0 ? `${resultaat.mislukt} mislukt.` : ''}`,
        link: `/campaigns/${campagne.id}`
      });
      
    } catch (err) {
      await updateCampagneStatus(campagne.id, 'gestopt');
      await logScheduler('verzending_fout', campagne.id, err.message);
    }
  }
}
```

### `planCampagne(campaignId)`
```javascript
// Berekent en slaat het werkelijke verzendmoment op

async function planCampagne(campaignId) {
  const campagne = await getCampagne(campaignId);
  
  if (campagne.verzend_direct) {
    // Direct verzenden: bereken toch correctie voor tijdvenster
    const nu = new Date();
    const werkelijkMoment = berekenWerkelijkVerzendMoment(
      nu,
      campagne.alleen_werkdagen,
      campagne.tijdvenster_start,
      campagne.tijdvenster_eind
    );
    
    // Als het gecorrigeerde moment meer dan 1 minuut in de toekomst ligt:
    // sla op als gepland en laat scheduler het oppakken
    // Anders: direct verzenden
    const verschilMs = werkelijkMoment - nu;
    
    if (verschilMs > 60000) {
      await updateCampagne(campaignId, {
        status: 'gepland',
        werkelijk_verzend_op: werkelijkMoment
      });
      return { uitgesteld: true, moment: werkelijkMoment };
    } else {
      return { uitgesteld: false };
    }
  } else {
    // Gepland moment: bereken correctie
    const werkelijkMoment = berekenWerkelijkVerzendMoment(
      new Date(campagne.verzend_op),
      campagne.alleen_werkdagen,
      campagne.tijdvenster_start,
      campagne.tijdvenster_eind
    );
    
    await updateCampagne(campaignId, {
      status: 'gepland',
      werkelijk_verzend_op: werkelijkMoment
    });
    
    return { uitgesteld: true, moment: werkelijkMoment };
  }
}
```

## 5. API endpoints — verzendplanning

### POST /api/campaigns/:id/plannen
```
Body: {
  verzend_direct: boolean,
  verzend_op: datetime | null,      -- null als verzend_direct = true
  alleen_werkdagen: boolean,
  tijdvenster_start: string,        -- bijv. "08:00"
  tijdvenster_eind: string          -- bijv. "18:00"

1. Valideer: als verzend_direct = false, verzend_op mag niet in het verleden zijn
2. Update campagne-instellingen
3. Roep planCampagne(campaignId) aan
4. Geef terug:
{
  werkelijk_verzend_op: datetime,
  verschil_melding: "Je mail is gepland op maandag 14 april om 09:00 
                     (verschoven van zaterdag 20:00)",
  gecorrigeerd: boolean  -- true als het moment verschoven is
}
```


## 11. Edge cases & foutafhandeling

Behandel de volgende situaties correct:

**Server herstart tijdens geplande campagne:**
- Bij server-opstart: voer `runAllSchedulers()` direct uit
- Campagnes met `werkelijk_verzend_op` in het verleden en status = 'gepland' worden direct opgepakt

**Campagne met 0 ontvangers:**
- Bij activatie of inplannen: controleer of segmentfilters minimaal 1 actief contact opleveren
- Zo niet: blokkeer activatie met melding "Geen actieve contacten gevonden voor dit segment"

**SendGrid rate limit:**
- Als SendGrid een 429 (Too Many Requests) teruggeeft:
  - Wacht 60 seconden en probeer opnieuw (max 3 pogingen)
  - Na 3 mislukte pogingen: markeer batch als mislukt, log fout, ga door met volgende batch
  - Update campagne status NIET naar 'voltooid' maar naar 'actief' met een fout-indicator

**Tijdzone correctie:**
- Alle berekeningen in `schedulerUtils.js` werken in de tijdzone 'Europe/Amsterdam'
- Gebruik de `Intl` API of een lichte tijdzone library voor correcte zomer/wintertijd afhandeling
- Sla alle timestamps op in UTC in de database, converteer naar NL tijd alleen voor weergave

**Dubbele verzending voorkomen:**
- Voeg een database-lock toe bij het starten van een campagneverzending:
  ```sql
  UPDATE campaigns SET status = 'actief' 
  WHERE id = ? AND status = 'gepland'
  ```
  Controleer of de update 1 rij heeft gewijzigd. Zo niet: een andere scheduler-instantie is al bezig, sla over.
